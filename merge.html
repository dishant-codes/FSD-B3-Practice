public void mergeSort(int[] arr, int l, int r) {
    if (l < r) {
        int mid = l + (r - l) / 2;
        mergeSort(arr, l, mid);     // Left half
        mergeSort(arr, mid + 1, r); // Right half
        merge(arr, l, mid, r);       // Combine
    }
}
